cmake_minimum_required(VERSION 3.22)
project(AETHER_ENGINE VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(engine/core)
include(CTest)
if(BUILD_TESTING)
    add_executable(aether_core_smoke tests/core_smoke.c)
    target_link_libraries(aether_core_smoke PRIVATE aether_core)
    add_test(NAME aether_core_smoke COMMAND aether_core_smoke)
endif()
